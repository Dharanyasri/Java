import java.util.Scanner;

class Node {
    int data;
    Node next;

    Node(int data) {
        this.data = data;
        this.next = null;
    }
}

public class Main {

    static Node head = null;

    public static void insertAtPosition(int data, int pos) {//100,1
        Node newNode = new Node(data);

        // Insert at beginning
        if (pos == 1) {//1==1
            newNode.next = head;
            head = newNode;
            return;
        }

        Node temp = head;//100

        // Move to (pos-1)th node
        for (int i = 1; i < pos - 1 && temp != null; i++) {//2<2
            temp = temp.next;//200
        }

        if (temp == null) {
            System.out.println("Invalid Position");
            return;
        }

        newNode.next = temp.next;
        temp.next = newNode;
    }

    public static void display() {
        Node temp = head;

        while (temp != null) {
            System.out.print(temp.data + " ");
            temp = temp.next;
        }
    }

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter number of nodes: ");
        int n = sc.nextInt();//3

        // Create the linked list
        for (int i = 1; i <= n; i++) {//4<=3
            System.out.print("Enter data: ");
            int data = sc.nextInt();//30

            insertAtPosition(data, i);//30 3
        }

        System.out.print("Enter data to insert: ");
        int data = sc.nextInt();//100

        System.out.print("Enter position: ");
        int pos = sc.nextInt();//1

        insertAtPosition(data, pos);

        System.out.println("Linked List:");
        display();
    }
}
