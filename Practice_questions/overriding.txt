class Vehicle{
    public void start(){
        System.out.println("Vehicle is starting");
    }
}
class Car extends Vehicle{
    public void start(){
        System.out.println("Car starts with a key");
    }
    public void display(){
        start();
        super.start();
    }
}
public class Main{
    public static void main(String args[]){
        Car c = new Car();
        c.display();
    }
}
