import java.util.*;
    class Main{
        public static void main(String args[]){
            Scanner sc = new Scanner(System.in);
            int n = sc.nextInt();
            int a[] = new int[n];
            
            for(int i=0;i<n;i++){
                a[i]=sc.nextInt();
            }
            
            HashSet<Integer> set = new HashSet();
            int ans = -1;
            for(int i=0;i<n;i++){
                if(set.contains(a[i])){
                    ans = a[i];
                    break;
                }
                set.add(a[i]);
            }
            System.out.println(ans);
        }
    }
