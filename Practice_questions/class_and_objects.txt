class Employee{
    String name;
    int salary;
    
    Employee(String name,int salary){
        this.name = name;
        this.salary=salary;
    }
    public void display(){
        
        System.out.println("Employee name:"+name);
        System.out.println("Employee salary:"+salary);
        
    }
}
public class Main{
    public static void main(String args[]){
        Employee e = new Employee("Rahul",40000);
         Employee e1 = new Employee("Jana",67000);
          Employee e2 = new Employee("Afra",80000);
          
          System.out.println("Employee 1 Details");
          e.display();
          System.out.println();
          System.out.println("Employee 2 Details");
          e1.display();
           System.out.println();
          System.out.println("Employee 3 Details");
          e2.display();
    }
}
