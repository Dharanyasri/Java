import java.util.*;
    class Main{
        public static void main(String args[]){
            Scanner sc = new Scanner(System.in);
            int n = sc.nextInt();
            
            Set<Integer> set1 = new HashSet<>();
            Set<Integer> set2 = new HashSet<>();
             Set<Integer> ans = new HashSet<>();
             
             for(int i=0;i<n;i++){
                 set1.add(sc.nextInt());
             }
             for(int i=0;i<n;i++){
                 set2.add(sc.nextInt());
             }
             for(int num:set1){
                 ans.add(num);
             }
             for(int num:set2){
                 ans.add(num);
             }
             System.out.print(ans);
        }
    }
